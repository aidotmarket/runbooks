   SQL inside the owner `psql -v ON_ERROR_STOP=1 "$DB_URL"` session, after the protected operator process has injected `CONNECTOR_RUNTIME_PASSWORD` into its environment (never argv, history or output), is:

   ```sql
   \getenv generated_password CONNECTOR_RUNTIME_PASSWORD
   BEGIN;
   CREATE ROLE connector_runtime LOGIN NOINHERIT NOSUPERUSER NOCREATEDB
     NOCREATEROLE NOREPLICATION NOBYPASSRLS PASSWORD :'generated_password';
   REVOKE ALL ON ALL TABLES IN SCHEMA public FROM connector_runtime;
   GRANT CONNECT ON DATABASE railway TO connector_runtime;
   GRANT USAGE ON SCHEMA public TO connector_runtime;
   GRANT INSERT, SELECT ON connector_audit_events TO connector_runtime;
   GRANT SELECT ON connector_switches, connector_oauth_grants,
     connector_oauth_clients, organization_memberships, organizations,
     users, seller_profiles, listings, notifications, data_requests
     TO connector_runtime;
   GRANT INSERT, UPDATE, SELECT ON connector_review_queue TO connector_runtime;
   COMMIT;
   ```
